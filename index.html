DAX Measures Library (`dax_measures.dax`)


[Overall Fit Score] = 
VAR SelectedCandidate = SELECTEDVALUE(Candidates[CandidateID])
RETURN
    IF(
        ISBLANK(SelectedCandidate),
        AVERAGE(AI_Recommendations[OverallFitScore]),
        CALCULATE(
            MAX(AI_Recommendations[OverallFitScore]),
            AI_Recommendations[CandidateID] = SelectedCandidate
        )
    )

[Skill Match %] = 
DIVIDE(
    COUNTROWS(CandidateSkills),
    COUNTROWS(RequiredSkills),
    0
)

[Candidate Rank] = 
RANKX(
    ALLSELECTED(Candidates),
    [Overall Fit Score],
    ,
    DESC,
    Dense
)

Power Query M Transformation (`power_query_transforms.m`)


let
    fnStandardizeSkill = (rawText as text) as text =>
    let
        cleanText = Text.Trim(Text.Lower(rawText)),
        standardized = 
            if cleanText = "python3" or cleanText = "python" then "Python"
            else if cleanText = "pbi" or cleanText = "powerbi" then "Power BI"
            else if cleanText = "sql server" or cleanText = "postgres" then "SQL"
            else Text.Proper(rawText)
    in
        standardized
in
    fnStandardizeSkill
